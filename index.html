function getEventPos(e) { let r = canvas.getBoundingClientRect(); // اصلاح باگ تاچ: در زمان touchend از changedTouches استفاده می‌شود تا آخرین نقطه لمس شده خوانده شود let touch = e.touches && e.touches.length > 0 ? e.touches[0] : (e.changedTouches ? e.changedTouches[0] : null); let clientX = touch ? touch.clientX : e.clientX; let clientY = touch ? touch.clientY : e.clientY; return { x: (clientX - r.left) * (canvas.width / r.width), y: (clientY - r.top) * (canvas.height / r.height) }; } function handleStart(e) { if (isMoving || shotsLeft <= 0) return; let pos = getEventPos(e); let d = Math.sqrt(Math.pow(pos.x - player.x, 2) + Math.pow(pos.y - player.y, 2)); // افزایش فضای لمس دور تیله بازیکن برای راحتی کار با انگشت روی موبایل if (d < player.r + 40) { isAiming = true; dragStart.x = pos.x; dragStart.y = pos.y; dragEnd.x = pos.x; dragEnd.y = pos.y; msgSlot.style.display = "none"; } } function handleMove(e) { if (!isAiming) return; let pos = getEventPos(e); dragEnd.x = pos.x; dragEnd.y = pos.y; } function handleEnd(e) { if (!isAiming) return; isAiming = false; // شلیک توپ با قدرتی متناسب با میزان کشیدن انگشت player.vx = (dragStart.x - dragEnd.x) * 0.11; player.vy = (dragStart.y - dragEnd.y) * 0.11; } // اتصال رویدادهای همزمان موس و تاچ برای پشتیبانی دسکتاپ و موبایل canvas.addEventListener("mousedown", handleStart); canvas.addEventListener("mousemove", handleMove); window.addEventListener("mouseup", handleEnd); canvas.addEventListener("touchstart", function(e) { e.preventDefault(); handleStart(e); }, { passive: false }); canvas.addEventListener("touchmove", function(e) { e.preventDefault(); handleMove(e); }, { passive: false }); window.addEventListener("touchend", function(e) { handleEnd(e); });
